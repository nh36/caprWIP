#!/usr/local/bin/foma -f
# Rule-based Proto-Celtic → daughter transducers.
#
# Daughters: Old_Irish, Middle_Welsh, Old_Welsh, Old_Breton, Middle_Breton,
#            Old_Cornish, Middle_Cornish, Gaulish, Celtiberian.
#
# Modelled on the structure of germanic.txt. Proto-Celtic segments use the
# {*x} brace notation throughout. Plain input characters are mapped to their
# starred counterparts by the pcelt* template definitions, and the starred
# forms are consumed by language-specific rule chains.
#
# Proto-Celtic consonantism: p, b, t, d, k, g, kʷ, gʷ, s, x, m, n, l, r, y,
#   w, ɸ.  No /f/ (labiodental) and no /ō/ (PIE *ō merged with *ā in PC);
#   both are daughter-language innovations or rule outputs.
#
# Phonological rules are simplified approximations. Complex conditioned changes
# (morphological lenition classes, OI epenthesis, Brythonic affection,
# Gaulish paradigm-specific apocope, etc.) produce known mismatches that are
# documented in the comparison report.

# 0a. initial *u before *ɸ behaves as *w in Insular Celtic
#     (*uɸo → fo/guo). Applied before OIInitialW / BritInitialW.
define PCeltInitialUPhi [ {*u} -> {*w} || .#. _ {*ɸ} ];

# 0b. *ɸ → 0 in all daughters.
define PCeltPhiLoss [ {*ɸ} -> 0 ];

## Old Irish transducer                     ##
##############################################

# 1. Labiovelars are Q-Celtic (merged with plain velars in Goidelic).
define OILabioVelar [
    {*kʷ} -> {*k},
    {*gʷ} -> {*g}
];

# 2. Initial *gd- cluster: *g drops (*gdonyos > duin).
define OIGdCluster [ {*g} {*d} -> {*d} || .#. _ ];

# 3. Initial *sf- cluster: *f drops (*sɸerā after phi-loss → *sera > ser).
define OISFCluster [ {*s} {*f} -> {*s} || .#. _ ];

# 4. Initial *w > *f (*wiros > fer, *windos > find).
define OIInitialW [ {*w} -> {*f} || .#. _ ];

# 5. *w after a consonant > 0 (*ardwos > ard).
define OIMedialWDrop [ {*w} -> 0 || PCeltStarConsonant _ ];

# 6. *w after a vowel > *u (*biwos > beo, *abū > auv).
define OIMedialWVocal [ {*w} -> {*u} || PCeltStarVowel _ ];

# 7a. *o before a consonant cluster ending in *y raises to *u (*koryos > kure,
#     *olyos > ule, *gdonyos > dune).
define OIORaising [ {*o} -> {*u} || _ PCeltStarConsonant+ {*y} ];

# 7b. *y after a consonant and before a vowel vocalises to *e (*aryos > are,
#     *koryos > kure). Must precede OIYDrop which drops all remaining *y.
define OIYVocal [ {*y} -> {*e} || PCeltStarConsonant _ PCeltStarVowel ];

# 7. *y > 0 everywhere in Old Irish (word-initial and vowel-flanked *y dropped).
define OIYDrop [ {*y} -> 0 ];

# 8. Nasal-anterior mutations. Word-final *ants > *a (drops *nts). Medial *ant
#    before a vowel > *ād (short) or *ēd (long); both paths kept for ambiguity.
#    Also *ent before a vowel > *ēd (compensatory lengthening: *sentus > sēd).
define OINasalAnt [
    {*a}{*n}{*t}{*s} -> {*a} || _ .#.,
    {*a}{*n}{*t} -> ({*ē} d | {*a} d) || _ PCeltStarVowel,
    {*a}{*n}{*k} -> ({*ē} g | {*a} g) || _ PCeltStarVowel,
    {*e}{*n}{*t} -> {*ē} d || _ PCeltStarVowel
];

# 9. *mb > *m (*kambos > cam).
define OIMbSimpl [ {*m} {*b} -> {*m} ];

# 9b. *nd > *n before long *ā (*tundā > ton, *landā > lan, *bandā > ban).
#     The *d in *Cndā clusters elides before the ā-suffix; this runs before
#     a-affection so that the *ā is still present to trigger lowering.
define OINdElision [ {*n}{*d} -> {*n} || _ {*ā} ];

# 9c. A-affection: high vowels *i, *u lower to mid *e, *o when the following
#     syllable contains a low/mid back vowel (*a, *ā, *o).  Non-deterministic
#     (both paths kept) to avoid over-commitment on ambiguous paradigm forms.
#     (*wiros > fer, *tundā > ton after OINdElision gives *tunā)
define OIAAffection [
    {*i} -> ({*e}|{*i}) || _ PCeltStarConsonant+ [{*a}|{*ā}|{*o}],
    {*u} -> ({*o}|{*u}) || _ PCeltStarConsonant+ [{*a}|{*ā}|{*o}]
];

# 9d. I-affection: mid vowels *e, *o raise to high *i, *u when the following
#     syllable contains a high vowel (*i, *ī, *u, *ū).  Non-deterministic.
#     (*meli > mil, *wolis > ful)
define OIIAffection [
    {*e} -> ({*i}|{*e}) || _ PCeltStarConsonant+ [{*i}|{*ī}|{*u}|{*ū}],
    {*o} -> ({*u}|{*o}) || _ PCeltStarConsonant+ [{*i}|{*ī}|{*u}|{*ū}]
];

# 10. *u > *o before *k + following vowel (*drukos > droch).
define OIULowering [ {*u} -> {*o} || _ PCeltStarConsonant {*o} ];

# 11. *s > 0 between vowels, with compensatory lengthening of preceding vowel
#     (*esoxs > ēo).
define OIMedialSLoss [
    {*a}{*s} -> {*ā} || _ PCeltStarVowel,
    {*e}{*s} -> {*ē} || _ PCeltStarVowel,
    {*i}{*s} -> {*ī} || _ PCeltStarVowel,
    {*o}{*s} -> {*ō} || _ PCeltStarVowel,
    {*u}{*s} -> {*ū} || _ PCeltStarVowel
];

# 12. *t > *θ between vowels (*latis > lath, *brātīr > brāθīr).
define OIMedialT [ {*t} -> {*θ} || PCeltStarVowel _ PCeltStarVowel ];

# 13. *d > *ð between vowels.
define OIMedialD [ {*d} -> {*ð} || PCeltStarVowel _ PCeltStarVowel ];

# 14. *k > *x between non-*a vowels (*drukos > droch, *ekʷos > ech).
#     Restricted to non-*a left context to preserve stops in forms like
#     *balkos > balk, *makʷos > mac.
define OIFinalK [
    {*k} -> {*x} || [PCeltStarVowel - {*a}] _ PCeltStarVowel
];

# 15. *b > *v between vowels (*abū > auv).
define OIBLenition [ {*b} -> {*v} || PCeltStarVowel _ PCeltStarVowel ];

# 16. Diphthong mergers in Old Irish.
define OIVowelDiphthongs [
    {*ei} -> {*ī},
    {*ai} -> {*e},
    {*au} -> {*o},
    {*ou} -> {*ū},
    {*ow} -> {*ū}
];

# 17. Nasal assimilation: *n > *ŋ before *g (*longā > loŋg).
#     Must run AFTER lenition (rules 11–15) and BEFORE *g spirantisation (18–20)
#     so *g is still present when *n assimilates. *ŋ is NOT in PCeltStarLenCtx,
#     so *ŋg clusters survive spirantisation intact.
define OINasalAssimN [ {*n} -> {*ŋ} || _ {*g} ];

# 18. *g > *ɣ between vowels, or after *r before a vowel (*lugus>luɣ,
#     *argantom>arɣad). *l+g (*bolgos>bolg) and *ŋ+g (*longā>loŋg) do NOT
#     spirantise — *l is excluded here, *ŋ is excluded by OINasalAssimN.
define OIMedialG [
    {*g} -> {*ɣ} || PCeltStarVowel _ PCeltStarVowel,
    {*g} -> {*ɣ} || {*r}          _ PCeltStarVowel
];

# 19. *g > *ɣ word-finally after any vowel.
define OIFinalG [ {*g} -> {*ɣ} || PCeltStarVowel _ .#. ];

# 20. *g > 0 word-finally after a long vowel (*rīgos→rī).
define OIGLossAfterLong [ {*g} -> 0 || PCeltStarLongVowel _ .#. ];

# 20b. *m > *v between vowels (*nemos > nev).
define OIMLenition [ {*m} -> {*v} || PCeltStarVowel _ PCeltStarVowel ];

# 20b. *ā shortens to *a before *u in diphthong position (*nāwā > nau, not nāu).
#      Runs after OIMedialWVocal creates *u from *w, before OIApocope drops *ā.
define OIAUShorten [ {*ā} -> {*a} || _ {*u} ];

# 21. Final-syllable apocope (four sequential passes).
#     Pass 3 no longer drops {*ī}: that was incorrectly stripping stem long-ī
#     from words like *trīs → trī (the *s drops in pass 2, leaving *ī final).
#     Pass 4 drops *-ns clusters after pass 3, so *mīns → mī without the *ī
#     being re-dropped (*ī is only final after pass 4, not pass 3).
define OIApocope [
    [{*o}{*s} -> 0, {*o}{*m} -> 0, {*a}{*m} -> 0, {*u}{*s} -> 0, {*x}{*s} -> 0 || _ .#.]
    .o.
    [{*s} -> 0 || PCeltStarVowel _ .#.]
    .o.
    [{*ā} -> 0, {*ū} -> 0, {*i} -> 0 || _ .#.]
    .o.
    [{*u} -> 0 || PCeltStarConsonant _ .#.]
    .o.
    [{*n}{*s} -> 0 || _ .#.]
];

# 21b. *ē before word-final *xs → *ē (with the *xs dropping via compensatory
#      lengthening). Runs before OIApocope (*swexs > sē via *sexs > *sē).
define OIXsLoss [
    {*e}{*x}{*s} -> {*ē} || _ .#.
];

# 21c. Word-final cluster simplification: *sn > *n (*wasnos > fan via *fasn > *fan).
define OIFinalCluster [
    {*s}{*n} -> {*n} || _ .#.
];

# 21d. Degemination after apocope (*slattā > slat, *mukkus > muk).
define OIDegemination [
    {*b}{*b} -> {*b}, {*d}{*d} -> {*d}, {*g}{*g} -> {*g},
    {*k}{*k} -> {*k}, {*l}{*l} -> {*l}, {*m}{*m} -> {*m},
    {*n}{*n} -> {*n}, {*p}{*p} -> {*p}, {*r}{*r} -> {*r},
    {*s}{*s} -> {*s}, {*t}{*t} -> {*t}
];

# 22. Strip asterisks; map long vowels to colon notation.
define OIRemoveStars [
    {*a} -> a,       {*ā} -> ā,
    {*e} -> e,       {*ē} -> ē,
    {*i} -> i,       {*ī} -> (ī | i),
    {*o} -> o,       {*ō} -> ō,
    {*u} -> u,       {*ū} -> ū,
    {*b} -> b,       {*d} -> d,
    {*f} -> f,       {*g} -> g,
    {*h} -> h,       {*k} -> k,
    {*l} -> l,       {*m} -> m,
    {*n} -> n,       {*p} -> p,
    {*r} -> r,       {*s} -> s,
    {*t} -> t,       {*v} -> v,
    {*w} -> u,       {*x} -> x,
    {*y} -> 0,
    {*θ} -> θ,       {*ð} -> ð,
    {*ɣ} -> ɣ,       {*χ} -> χ,
    {*ŋ} -> ŋ
];

# Early OI breaking rules (applied before OINasalAnt to preserve OINasalAnt-
# generated *ē and before OIIWDiphthong to avoid breaking new *ē from *iw).
# 4a. *ē > *ī*a in Old Irish (*mēnom > mīan).
define OIEBreaking [ {*ē} -> {*ī}{*a} ];

# 4b. *i*w > *ē*u (*biwos > bēu). Must precede OIMedialWDrop/OIMedialWVocal.
define OIIWDiphthong [ {*i}{*w} -> {*ē}{*u} ];

# 16b. Compensatory lengthening: stop/obstruent absorbed before *l or *n

    {*e}{*t} -> {*ē} || _ [{*l}|{*n}],
    {*o}{*t} -> {*ō} || _ [{*l}|{*n}],

# 1. *kʷ > *p, *gʷ > *b (the defining P-Celtic innovation).
define BritLabioVelar [
    {*kʷ} -> {*p},
    {*gʷ} -> {*b}
];

# 2. Initial *gd- > *d (*gdonyos > din).
define BritGdCluster [ {*g} {*d} -> {*d} || .#. _ ];

# 2b. Initial *skʷ- > *χw- BEFORE BritLabioVelar converts *kʷ→*p,
#     so *skʷetlom → *χwetlom (MW chwedl) not *spetlom (OW).
# The *w is unstarred so BritMedialWDrop (which only matches {*w}) cannot delete it.
define BritSkwCluster [ {*s} {*kʷ} -> {*χ} w || .#. _ ];

# 3. Initial consonant-cluster simplifications before other initial-s rules.
#    *sr- > *fr- (shared with Gaulish; *srognā > froin/frogna).
#    *sw- > *χw- (MW chw-), *sf- > *f-, *sɸ- > *f- (*sɸerā > fer), *sm- > *m-, *sl- > *l-.
define BritInitialSClusters [
    {*s} {*r} -> {*f} {*r} || .#. _,
    {*s} {*w} -> {*χ} w    || .#. _,
    {*s} {*f} -> {*f}       || .#. _,
    {*s} {*ɸ} -> {*f}       || .#. _,
    {*s} {*m} -> {*m}       || .#. _,
    {*s} {*l} -> {*l}       || .#. _
];

# 4. Remaining initial *s > *h.
define BritInitialS [ {*s} -> {*h} || .#. _ ];

# 5. Initial *w > *gw (*wiros > gwr, *windos > gwinn).
#    gw appears as g+u in the IPA tokens (OW guo, MW guið etc.).
define BritInitialW [ {*w} -> {*gw} || .#. _ ];

# 6. *w after a consonant > 0 (*ardwos > arð).
define BritMedialWDrop [ {*w} -> 0 || PCeltStarConsonant _ ];

# 7. *ye > *ia word-initially (Wikipedia §5.2.2.4: *ye > /ja/; *yegis > Welsh iâ "ice").
#    *y > *i word-initially before other vowels (MW iant/yantus, iaul/yālos).
#    *y > *ð OR 0 when it will be word-final after apocope of *-os/*-Cs
#      (OW korð ← *koryos, MW ol ← *olyos; Wikipedia: *-j > *-ð "or lost").
#    *y > 0 elsewhere (medial *y in *smiyoros; initial *y before consonant).
define BritYDrop [
    {*y}{*e} -> {*i}{*a} || .#. _,
    {*y} -> {*i} || .#. _ PCeltStarVowel,
    {*y} -> ({*ð} | 0) || PCeltStarConsonant _ PCeltStarShortVowel PCeltStarConsonant .#.,
    {*y} -> 0
];

# 8. *ū > *i (short i; Brythonic u-raising: *dūnom > din, *rūnā > rin).
#    Produces short surface i, not long ī, matching all TSV daughter forms.
define BritUtoI [ {*ū} -> {*i} ];

# 9. *mb > *m (*kambos > camm/cam in Brythonic).
define BritMbSimpl [ {*m} {*b} -> {*m} ];

# 9b. Geminate simplification with spirantisation/fricativisation.
#     *tt → *θ (surface: OW/OCorn th, OBret z/h).
#     *kk → *x (then BritFinalX spirantises to *χ word-finally; vowel often lengthens).
#     *ll, *nn, *mm → single consonants.
define BritGeminates [
    {*t}{*t} -> {*θ},
    {*k}{*k} -> {*x},
    {*l}{*l} -> {*l},
    {*n}{*n} -> {*n},
    {*m}{*m} -> {*m}
];

# 10. Diphthong mergers in Brythonic.
#     *au > *u (OW bu from *baus); *ow > *aw or *u (naw/*rud from *nowan/*rowdos).
define BritVowelDiphthongs [
    {*ei} -> {*oi},
    {*ai} -> {*oi},
    {*au} -> {*u},
    {*ou} -> {*u},
    {*ow} -> ({*aw} | {*u})
];

# 10b. I-affection (umlaut): *a → *i in syllable before *ī (e.g. *bleidanī > bloiðin).
#      Applied before apocope so the conditioning *ī is still present.
define BritIAffection [ {*a} {*n} {*ī} -> {*i} {*n} {*ī} ];

# 10b3. *o-affection (i-affection of *o): *o → *e in syllable before consonant(s) + *i.
#       Applied after BritIAffection so *a → *i affection fires first.
#       Example: *wolis > *gwoli > *gweli > gueli.
define BritOIAffection [ {*o} -> {*e} || _ PCeltStarConsonant+ {*i} ];

# 10b2. *xt → *iθ: *x vocalises before *t, spirantising *t and creating a glide.
#        Applied after BritUtoI (so *ū is already *i) and before BritApocope.
#        Example: *oxtū → *ox+ti (UtoI) → *o+iθ+i (XTCluster) → *oiθ (apocope drops *i).
#        OW/OCorn surface: "oiθ"; OBret surface: *xt → iθ but preceding *o→e (via RemoveStars).
define BritXTCluster [ {*x}{*t} -> {*i}{*θ} ];

# 10c. A-affection: *u → *o, *i → *e in the syllable before a final *ā (Wikipedia §5.2.2.7).
#      Applied before apocope so the conditioning *ā is still present.
#      (*tundā > *tondā > ton; feminine *windā > gwen vs masculine *windos > gwyn).
define BritAAffection [
    {*u} -> {*o} || _ PCeltStarConsonant+ {*ā},
    {*i} -> {*e} || _ PCeltStarConsonant+ {*ā}
];

# 10d. Assimilation of *oRa to *aRa (Wikipedia §5.2.2.3; R = lone sonorant).
#      Unlike Joseph's rule, this change did not occur in Goidelic.
#      (*toranos > *taranos > taran; must precede *mVn- raisings and a-affection).
define BritORAtoARA [ {*o} -> {*a} || _ PCeltStarSonorant [{*a}|{*ā}] ];

# 10e. Nasal assimilation: *n > *m before *p (from labiovelar *kʷ > *p).
#      (*kʷenkʷe > *penpe > *pempe > pimp/pemp).
define BritNasalAssim [ {*n}{*p} -> {*m}{*p} ];

# 10f. Raisings: *e > *i (non-deterministic) before nasal+stop (Wikipedia §5.2.2.1).
#      Covers *sentus > hint/*hins, *kʷenkʷe > pimp/pemp.
#      Non-deterministic to allow dialect variation (MW pimp, OBret pemp).
define BritRaising [
    {*e} -> ({*i}|{*e}) || _ [{*n}|{*m}] [{*t}|{*d}|{*k}|{*g}|{*p}|{*b}]
];

# 10g. *ei merger before nasals: *ei → *e before *n/*m.
#      *saleinos > halen/*holen: the *ei diphthong simplifies to *e before word-final nasal
#      (otherwise BritVowelDiphthongs converts *ei → *oi, giving halin not halen).
#      Applied BEFORE BritVowelDiphthongs.
define BritEINasalMerge [ {*ei} -> {*e} || _ [{*n}|{*m}] ];

# 11. Final-syllable apocope (mirrors OIApocope; pass 2 drops *s only).
#     Pass 3 does NOT drop *o: *uɸo → guo (final *o is stem, not paradigm ending).
define BritApocope [
    # retain final *us to allow reconstruction (e.g. *bitus > bid)
    [{*o}{*s} -> 0, {*o}{*m} -> 0 || _ .#.]
    .o.
    [{*s} -> 0 || PCeltStarVowel _ .#.]
    .o.
    [{*t}{*u} -> {*t} || _ .#.]
    .o.
    [{*ā} -> 0, {*i} -> 0 || _ .#.]
    .o.
    # Drop *ī word-finally only in multi-syllabic forms (V C+ before *ī),
    # so *bloidinī > bloiðin but monosyllabic *trī (< *trīs) keeps its *ī.
    [{*ī} -> 0 || PCeltStarVowel PCeltStarConsonant+ _ .#.]
    .o.
    # *xs > *χ word-finally (MW χwēχ from *swexs; *natrixs > natriχ).
    [{*x}{*s} -> {*χ} || _ .#.]
    .o.
    # *ns > *s word-finally (*mīns > mīs, *karants partial).
    [{*n}{*s} -> {*s} || _ .#.]
    .o.
    # *ants > 0 word-finally (*karants > kar, *tarants, *karant...).
    [{*a}{*n}{*t}{*s} -> 0 || _ .#.]
    .o.
    # Drop final *m after *ā (e.g. *ɸlāmā > *lāmā > *lām > *lā > lau).
    [{*m} -> 0 || {*ā} _ .#.]
    .o.
    # Drop *e word-finally after a consonant (numeral apocope: *kʷenkʷe > pemp/pimp).
    [{*e} -> 0 || PCeltStarConsonant _ .#.]
];

# 12. Lenition of obstruents in weak position. Each rule is a separate define
#     composed sequentially to avoid parallel-replacement + .#. interactions.
# Medial *xs → *χ before vowel (*ouxselos > uχel).
define BritMedialXS [ {*x}{*s} -> {*χ} || _ PCeltStarVowel ];
# *sn cluster simplification (*kʷresnom > pren; *s drops before nasal).
define BritSNSimpl [ {*s}{*n} -> {*n} ];
# *ī > 0 before word-final *r (r-stem syncope: *brātīr > brotr).
# Only syncopates *ī in the r-stem suffix V-C-ī-r (kinship terms: *mātīr, *brātīr etc.)
# A bare *C-ī-r stem like *tīros→*tīr must NOT be affected.
define BritIRSyncope [ {*ī} -> 0 || PCeltStarVowel PCeltStarConsonant _ {*r} .#. ];
define BritLenInterD [ {*d} -> {*ð} || PCeltStarLenCtx _ PCeltStarVowel ];
define BritLenFinalD [ {*d} -> {*ð} || PCeltStarLenCtx _ .#. ];
define BritLenT     [ {*t} -> {*d} || PCeltStarVowel _ PCeltStarSonorant ];
define BritLenFinalK  [ {*k} -> {*χ} || PCeltStarSonorant _ .#. ];
# Final *k → *g after vowels (e.g. *drukos → drwg, *mrakis → brag).
define BritVowelFinalK [ {*k} -> {*g} || PCeltStarVowel _ .#. ];
define BritLenInterG [ {*g} -> {*ɣ} || PCeltStarVowel _ PCeltStarVowel ];
# Final *g → *ɣ after vowel before sonorant (*maglos > mail, *tigernos > teyrn).
# The *ɣ then surfaces as i (glide) via BritRemoveStars, creating diphthongs.
define BritLenGBeforeSon [ {*g} -> {*ɣ} || PCeltStarVowel _ PCeltStarSonorant .#. ];
# Final *t → *d only after SHORT vowels (e.g. *bitus → bid, not *brātus → braud).
define BritLenFinalT [ {*t} -> {*d} || PCeltStarVowel _ .#. ];
# Final *t → *θ after *r or *l (e.g. *nertom → nerθ, *slattā → laθ).
define BritFinalTSpirRL [ {*t} -> {*θ} || [{*r} | {*l}] _ .#. ];
# Final *nd → *n: word-final *d after nasal is lost (e.g. *sindos → hin, *landā → lan).
define BritFinalNDrop [ {*n} {*d} -> {*n} || _ .#. ];
# *e → *i before *g: vowel raising before velars (e.g. *tegos → tig in Old Welsh).
define BritEtoIBeforeG [ {*e} -> {*i} || _ {*g} ];
# Final *ng > *ŋ: velar nasal coalescence at word end (*longā > loŋg/ɬoŋ).
define BritFinalNGCoal [ {*n} {*g} -> {*ŋ} || _ .#. ];
# *x → *χ word-finally after a vowel (spirantisation; follows geminate *kk→*x).
# E.g. *mukkus → *mux → *muχ → OW/OCorn moχ; also handles *oxtū path.
define BritFinalX [ {*x} -> {*χ} || PCeltStarVowel _ .#. ];
define BritLenition [
    BritSNSimpl
    .o. BritIRSyncope
    .o. BritFinalX
    .o. BritMedialXS .o. BritFinalNGCoal .o. BritFinalNDrop .o. BritEtoIBeforeG
    .o. BritLenInterD .o. BritLenFinalD .o. BritLenT .o. BritFinalTSpirRL .o. BritLenFinalT .o. BritLenFinalK .o. BritVowelFinalK .o. BritLenInterG .o. BritLenGBeforeSon
];


define BritRemoveStars [
    {*a}  -> a,      {*ā}  -> (ā | {au} | a | o),
    {*e}  -> (e | ē),  {*ē}  -> ē,
    {*i}  -> i,      {*ī}  -> (ī | i),
    {*o}  -> o,      {*ū}  -> ū,
    {*u}  -> (u | ū | o),
    {*oi} -> ({oi} | i | {ui}),   {*aw} -> {aw},
    {*b}  -> b,      {*d}  -> (d | t),
    {*f}  -> f,      {*g}  -> (g | ɣ | 0),
    {*gw} -> (g u | g ū),    {*h}  -> h,
    {*k}  -> k,      {*l}  -> (l | ɬ),
    {*m}  -> m,      {*n}  -> n,
    {*p}  -> (p | b),  {*r}  -> r,
    {*s}  -> s,      {*t}  -> t,
    {*v}  -> v,      {*w}  -> (u | w),
    {*x}  -> x,      {*y}  -> 0,
    {*θ}  -> θ,      {*ð}  -> (ð | d),
    {*ɣ}  -> (ɣ | 0 | i),  {*χ}  -> (χ | 0),
    {*ŋ}  -> ŋ
];

# Old Breton / Middle Breton surface mappings.
# Key differences from Welsh:
#   *ā → e/eu (Breton lost the Welsh aw/au diphthong)
#   *i, *ī → e (short *i merged with *e in Breton)
#   *o → e (Breton raised *o in several environments, e.g. *gdonyos → den, *oiθ → eiθ)
#   *u → o (Brythonic*u → o in Breton/Cornish)
#   *ū → o/u (long merged with short before becoming o)
#   *θ → z (Breton realises the dental spirant as z, e.g. *rextus → rez)
#   *ð → z or lost (lenited *d → z or drops in Breton)
#   *χ → h (Breton realises the velar fricative as h in some positions)
#   *w → ū (word-initial *w sequences can surface as ū in OBret, e.g. *nowan → naū)
#   *l → l (Breton did not develop voiceless ɬ)
define OBretRemoveStars [
    {*a}  -> a,           {*ā}  -> (e | {eu} | a | o),
    {*e}  -> (e | ē),     {*ē}  -> ē,
    {*i}  -> (e | i),     {*ī}  -> (e | ī | i),
    {*o}  -> (o | e),     {*ū}  -> (u | o | ū),
    {*u}  -> (o | u | ū),
    {*oi} -> {oe},        {*aw} -> {aw},
    {*b}  -> b,           {*d}  -> (d | t),
    {*f}  -> f,           {*g}  -> (g | ɣ | 0),
    {*gw} -> (g u | g ū), {*h}  -> h,
    {*k}  -> k,           {*l}  -> l,
    {*m}  -> m,           {*n}  -> n,
    {*p}  -> (p | b),     {*r}  -> r,
    {*s}  -> s,           {*t}  -> t,
    {*v}  -> v,           {*w}  -> (u | ū | w),
    {*x}  -> x,           {*y}  -> 0,
    {*θ}  -> (z | θ),     {*ð}  -> (z | d | 0),
    {*ɣ}  -> (ɣ | 0 | i), {*χ}  -> (χ | 0 | h),
    {*ŋ}  -> ŋ
];

# Old Cornish / Middle Cornish surface mappings.
# Key differences from Welsh:
#   *ā → ue/u (Cornish *ā → ue or u)
#   *i → e (short *i → e in Cornish)
#   *u → o (Byrthonic *u → o in Cornish)
#   *ū → o/u (as in Breton)
#   *l → l/ɬ (Cornish preserved voiceless ɬ in some positions)
#   *θ, *ð stay as θ, ð (Cornish did not shift to z like Breton)
define OCornRemoveStars [
    {*a}  -> a,           {*ā}  -> ({ue} | u | e | a | o),
    {*e}  -> (e | ē),     {*ē}  -> ē,
    {*i}  -> (e | i),     {*ī}  -> (ī | i | e),
    {*o}  -> o,           {*ū}  -> (u | o | ū),
    {*u}  -> (o | u | ū),
    {*oi} -> {oi},        {*aw} -> {aw},
    {*b}  -> b,           {*d}  -> (d | t),
    {*f}  -> f,           {*g}  -> (g | ɣ | 0),
    {*gw} -> (g u | g ū),  {*h}  -> h,
    {*k}  -> k,           {*l}  -> (l | ɬ),
    {*m}  -> m,           {*n}  -> n,
    {*p}  -> (p | b),     {*r}  -> r,
    {*s}  -> s,           {*t}  -> t,
    {*v}  -> v,           {*w}  -> (u | w),
    {*x}  -> x,           {*y}  -> 0,
    {*θ}  -> θ,           {*ð}  -> (ð | d),
    {*ɣ}  -> (ɣ | 0 | i), {*χ}  -> (χ | 0),
    {*ŋ}  -> ŋ
];

##############################################
## Proto-Brythonic transducer               ##
# Shared sound changes from Proto-Celtic to Proto-Brythonic.
# All starred segments throughout; no surface removal applied here.
# This chain is split into two stages so that daughter-language apocope rules
# (e.g. Middle Welsh *-am loss) can be inserted between apocope and lenition.

# Stage 1: everything up to (but not including) lenition.
define ProtoBrythPreLenRules [
    PCeltInitialUPhi
    .o. BritSkwCluster
    .o. BritLabioVelar
    .o. BritNasalAssim
    .o. BritGdCluster
    .o. BritInitialSClusters
    .o. PCeltPhiLoss
    .o. BritInitialS
    .o. BritInitialW
    .o. BritMedialWDrop
    .o. BritYDrop
    .o. BritMbSimpl
    .o. BritGeminates
    .o. BritEINasalMerge
    .o. BritVowelDiphthongs
    .o. BritORAtoARA
    .o. BritRaising
    .o. BritAAffection
    .o. BritOIAffection
    .o. BritUtoI
    .o. BritIAffection
    .o. BritXTCluster
    .o. BritApocope
];

# Stage 2: lenition (shared across all Brythonic, happened before the
# Welsh / Breton / Cornish dialect split).
define ProtoBrythLenition BritLenition;

# Complete Proto-Brythonic rule chain (starred → starred).
define ProtoBrythRules ProtoBrythPreLenRules .o. ProtoBrythLenition;

# ProtoBrythonic transducer: maps Proto-Celtic input to Proto-Brythonic
# intermediate representation (still starred; no surface form yet).
define ProtoBrythonic CelticProtoInput .o. ProtoBrythRules;

##############################################
## Welsh branch                             ##
##############################################

# --- Old Welsh ---
# No OW-specific phonological rules are needed beyond the shared Proto-Brythonic
# changes at present.  This is the hook for future OW innovations (e.g. vowel
# affection details, specific cluster changes).
define OldWelsh ProtoBrythonic .o. BritRemoveStars .o. BritSurface;

# --- Middle Welsh ---
# MW builds on Old Welsh by adding changes that took place after the OW period.
#
# MW1. Final *-am apocope: word-final *am → 0.
#      This MUST be inserted between ProtoBrythPreLenRules and ProtoBrythLenition
#      so that consonants newly word-final after *-am loss are lenited correctly
#      (e.g. *sextam: *t newly word-final after *am drops → θ via spirantisation).
define MWAmApocope [ {*a}{*m} -> 0 || _ .#. ];

# MW2. Final *r deletion after *d: *Vdr → *Vd at word end.
#      OW/OBret keep *r (e.g. *mātīr > OW modr, *brātīr > OBret brotr),
#      but MW loses it (e.g. *brātīr > MW braud).
define MWFinalRAfterD [ {*r} -> 0 || {*d} _ .#. ];

# MW3. Final *u deletion after consonant: *Cu → 0 at word end.
#      Extends apocope of high back vowel word-finally.
#      Example: *skātu > isgāt > isgād (with BritLenFinalT) > isgād.
#               *weidus > gueið (combined with *oi→i in RemoveStars).
define MWFinalU [ {*u} -> 0 || PCeltStarConsonant _ .#. ];

# MW4. Raising of *e to *i after *w before final *l: *wel → *wil at word end.
#      Covers *swelos > χwil (chwyl): BritInitialSClusters converts *sw → *χ + w (unstarred),
#      so the context is unstarred w (not {*w}).  Applied after lenition.
define MWEtoIAfterWBeforeL [ {*e} -> {*i} || w _ {*l} .#. ];

# MW rule chain: shared pre-lenition changes, then MW apocope, then lenition,
# then MW-specific post-lenition changes.
define MWStarred CelticProtoInput
    .o. ProtoBrythPreLenRules
    .o. MWAmApocope
    .o. ProtoBrythLenition
    .o. MWFinalRAfterD
    .o. MWFinalU
    .o. MWEtoIAfterWBeforeL;

define MiddleWelsh MWStarred .o. BritRemoveStars .o. BritSurface;

## Other Brythonic (Breton, Cornish)        ##

define OBretReflexes  CelticProtoInput .o. ProtoBrythRules .o. OBretRemoveStars;

##############################################
## Gaulish transducer                        ##
##############################################

# Gaulish is a Continental P-Celtic language that preserves endings more
# fully than Insular Celtic. Whether a particular form retains its final
# syllable depends on the declension and register; no general apocope rule
# is applied here — such forms will mismatch the TSV rather than
# over-deleting correct endings.

# 0g. Gaulish-specific treatment of initial *u before *ɸ: *u → *v, not *w.
#     (*uɸo > vo, *uɸer > ver). Applied before PCeltPhiLoss.
define GaulInitialUPhi [ {*u} -> {*v} || .#. _ {*ɸ} ];

# 1. *kʷ > *p, *gʷ > *b (P-Celtic).
define GaulLabioVelar [
    {*kʷ} -> {*p},
    {*gʷ} -> {*b}
];

# 2. Initial *sr- > *fr- (attested in *srognā > frogna; shared with Brythonic).
define GaulInitialSR [ {*s} {*r} -> {*f} {*r} || .#. _ ];

# 3. *w > *u everywhere (*wiros > uiros, *windos > uindus).
define GaulWVocal [ {*w} -> {*u} ];

# 4. *y > 0.
define GaulYDrop [ {*y} -> 0 ];

# 5. Diphthong reductions.
define GaulVowelDiphthongs [
    {*ei} -> {*e},
    {*ai} -> {*e},
    {*au} -> {*o},
    {*ou} -> {*u},
    {*ow} -> {*u}
];

# 6. Strip asterisks; macron notation for Gaulish long vowels.
define GaulRemoveStars [
    {*a}  -> a,   {*ā}  -> ā,
    {*e}  -> e,   {*ē}  -> ē,
    {*i}  -> i,   {*ī}  -> ī,
    {*o}  -> o,   {*ū}  -> ū,
    {*u}  -> u,
    {*b}  -> b,   {*d}  -> d,
    {*f}  -> f,   {*g}  -> g,
    {*k}  -> k,   {*l}  -> l,
    {*m}  -> m,   {*n}  -> n,
    {*p}  -> p,   {*r}  -> r,
    {*s}  -> s,   {*t}  -> t,
    {*v}  -> v,   {*w}  -> u,
    {*x}  -> x,   {*y}  -> 0,
    {*ŋ}  -> ŋ,   {*ɣ}  -> ɣ
];

define GaulRules [
    GaulInitialUPhi
    .o. PCeltPhiLoss
    .o. GaulLabioVelar
    .o. GaulInitialSR
    .o. GaulWVocal
    .o. GaulYDrop
    .o. GaulVowelDiphthongs
    .o. GaulRemoveStars
];

define GaulReflexes CelticProtoInput .o. GaulRules;
define Gaulish      GaulReflexes .o. GaulSurface;

## Celtiberian transducer                    ##
##############################################

# Celtiberian is a Q-Celtic Continental language. The TSV data (11 cognate
# sets) show conservative preservation with no length marking.

# 1. *kʷ > k+u, *gʷ > g+u (Q-Celtic labiovelars preserved as clusters).
define CeltibLabioVelar [
    {*kʷ} -> {*k} {*u},
    {*gʷ} -> {*g} {*u}
];

# 2. *w > *u everywhere.
define CeltibWVocal [ {*w} -> {*u} ];

# 3. Initial *d > *t (devoicing attested in *deiwos > teiuo).
define CeltibInitialD [ {*d} -> {*t} || .#. _ ];

# 4. *x > 0 (*rextus > retu).
define CeltibXLoss [ {*x} -> 0 ];

# 5. *y > 0.
define CeltibYDrop [ {*y} -> 0 ];

# 5a. Diphthong reductions (*au > *o as in *baus > bo).
define CeltibDiphthongs [ {*au} -> {*o}, {*ou} -> {*u} ];

# 5b. Final apocope: drop *s after vowel (*lugus > lugu, *deiwos > teiuo).
define CeltibApocope [ {*s} -> 0 || PCeltStarVowel _ .#. ];

# 6.
define CeltibRemoveStars [
    {*a}  -> a,   {*ā}  -> a,
    {*e}  -> e,   {*ē}  -> e,
    {*i}  -> i,   {*ī}  -> i,
    {*o}  -> o,
    {*u}  -> u,   {*ū}  -> u,
    {*b}  -> b,   {*d}  -> d,
    {*g}  -> g,   {*k}  -> k,
    {*l}  -> l,   {*m}  -> m,
    {*n}  -> n,   {*p}  -> p,
    {*r}  -> r,   {*s}  -> s,
    {*t}  -> t,   {*v}  -> v,
    {*w}  -> u,   {*x}  -> 0,
    {*y}  -> 0,   {*ŋ}  -> n,
    {*ei} -> e i, {*ai} -> a i,
    {*au} -> a u, {*ou} -> o u,
    {*ow} -> o u
];

define CeltibRules [
    PCeltPhiLoss
    .o. CeltibLabioVelar
    .o. CeltibWVocal
    .o. CeltibInitialD
    .o. CeltibXLoss
    .o. CeltibYDrop
    .o. CeltibDiphthongs
    .o. CeltibApocope
    .o. CeltibRemoveStars
];

define CeltibReflexes CelticProtoInput .o. CeltibRules;
define Celtiberian    CeltibReflexes .o. CeltibSurface;
