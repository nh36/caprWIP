# Two-level transducer that wraps German IPA symbols in braces.

    "aː":{aː} |
    "eː":{eː} |
    "iː":{iː} |
    "oː":{oː} |
    "uː":{uː} |

    a:{a} |
    b:{b} |
    d:{d} |
    e:{ɛ} |
    f:{f} |
    g:{ɡ} |
    h:{h} |
    i:{ɪ} |
    j:{j} |
    k:{k} |
    l:{l} |
    m:{m} |
    n:{n} |
    o:{ɔ} |
    p:{p} |
    r:{r} |
    s:{s} |
    t:{t} |
    u:{ʊ} |
    v:{v} |
    w:{w} |
    x:{x} |
    y:{y} |
    z:{z} |
    "ç":{ç} |
    "ʃ":{ʃ} |
    "ŋ":{ŋ} |
    "ɡ":{ɡ} |
    "ʊ":{ʊ} |
    "ɔ":{ɔ} |
    "ə":{ə} |
    "ɛ":{ɛ} |
    "ɪ":{ɪ} |
    "ː":{ː}
];

define GermanSurfacePrep GermanSurfacePrepSymbol*;
regex GermanSurfacePrep;
save stack german_surface_prep.bin
