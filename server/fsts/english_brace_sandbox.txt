# English sandbox: copy of current pipeline for brace migration experiments.

define EnglishSandboxSurfaceVowel [
    {a} | {e} | {i} | {o} | {u} | {y} |
    {aɪ} | {aʊ} | {eɪ} | {iː} | {uː} | {æ} | {ɑː} | {ɔ} | {ɔː} |
    {ə} | {əʊ} | {ɛ} | {ɛə} | {ɜː} | {ɪ} | {ɪə} | {ʊ} | {ʌ} |
    {ā} | {ē} | {ī} | {ō} | {ū} | {ą} | {ę}
];
define EnglishSandboxSurfaceConsonant [
    {b} | {d} | {f} | {g} | {h} | {j} | {k} | {l} | {m} | {n} | {p} |
    {r} | {s} | {t} | {v} | {w} | {z} | {ð} | {ŋ} | {ɡ} | {ʃ} | {ʋ} |
    {ʤ} | {ʧ} | {θ}
];
define EnglishSandboxSurfaceOnset [
    0 |
    EnglishSandboxSurfaceConsonant |
    EnglishSandboxSurfaceConsonant EnglishSandboxSurfaceConsonant |
    EnglishSandboxSurfaceConsonant EnglishSandboxSurfaceConsonant EnglishSandboxSurfaceConsonant
];
define EnglishSandboxSurfaceCoda [
    0 |
    EnglishSandboxSurfaceConsonant |
    EnglishSandboxSurfaceConsonant EnglishSandboxSurfaceConsonant |
    EnglishSandboxSurfaceConsonant EnglishSandboxSurfaceConsonant EnglishSandboxSurfaceConsonant
];
define EnglishSandboxSurfaceSyllable EnglishSandboxSurfaceOnset EnglishSandboxSurfaceVowel EnglishSandboxSurfaceCoda;
define EnglishSandboxSurface EnglishSandboxSurfaceSyllable+;

# Helper sets for the plain (post-brace) stages.
define EnglishSandboxPlainVocalic [
    EnglishSandboxSurfaceVowel |
    {e} | {o} | {u} | {oː}
];
define EnglishSandboxPlainLiquid [{l} | {r}];
define EnglishSandboxPlainNasal [{m} | {n}];

# Brace-aware helper inventories for the sandbox.
define EnglishSandboxStarVowel GermanStarVowel;

define EnglishSandboxStarDiphthong GermanStarDiphthong;

define EnglishSandboxStarConsonant GermanStarConsonant;

define EnglishSandboxStarVocalic GermanStarVocalic;

define EnglishSandboxStarNasal [{*m} | {*n}];

define EnglishSandboxStarLiquid [{*l} | {*r}];

define EnglishSandboxStarVelarStop [{*k} | {*g}];

define EnglishSandboxWeakTailVowel pgrmWeakTailVowel.r;

define EnglishSandboxInitialKnMarkers [
    {*k} {*n} -> {Kn} || .#. _,
    {*g} {*n} -> {Gn} || .#. _
];

define EnglishSandboxWGlideRule [{*w} -> {ʋ}];

define EnglishSandboxZRhotacism [
    {*z} -> {r} || EnglishSandboxStarVocalic _ EnglishSandboxStarVocalic
];

define EnglishSandboxZApocope [{*z} -> 0 || _ .#.];

define EnglishSandboxDJPalatal [{*d} {*j} -> {nd}];

define EnglishSandboxConsonantRules EnglishSandboxWGlideRule
    .o. EnglishSandboxZRhotacism
    .o. EnglishSandboxZApocope
    .o. EnglishSandboxDJPalatal;

define EnglishSandboxGhMarker [
    {*x} -> {GH} || EnglishSandboxStarVocalic _ ({*t} | {*d} | .#.)
];

define EnglishSandboxSilentInitialCleanup [
    "Kn" -> "n",
    "Gn" -> "n"
];

define EnglishSandboxGlideDeletion [
    {ʋ} -> 0 || EnglishSandboxStarVocalic _ EnglishSandboxStarVocalic
];

define EnglishSandboxGhDeletion [
    {GH} -> 0 || EnglishSandboxStarVocalic _ ({*t} | {*d} | .#.)
];

# West Germanic monophthongisation (ai → ā, au → ō).
define EnglishSandboxWestGermanic [
    {*ai} -> {*ā},
    {*au} -> {*ō}
];

# Early Old English open-syllable lengthening promotes short vowels before CV onsets.
define EnglishSandboxOpenSyllableLengthening [
    {*a} -> {*ā} || _ EnglishSandboxStarConsonant EnglishSandboxStarVocalic,
    {*e} -> {*ē} || _ EnglishSandboxStarConsonant EnglishSandboxStarVocalic,
    {*i} -> {*ī} || _ EnglishSandboxStarConsonant EnglishSandboxStarVocalic,
    {*o} -> {*ō} || _ EnglishSandboxStarConsonant EnglishSandboxStarVocalic,
    {*u} -> {*ū} || _ EnglishSandboxStarConsonant EnglishSandboxStarVocalic
];

# Anglo-Frisian breaking / rounding before liquids, /r/, and /w/ glides.
define EnglishSandboxBreakingLengthening [
    {*a} -> {*ō} || _ {*l} EnglishSandboxStarConsonant,
    {*a} -> {*ō} || _ {*l} .#.,
    {*a} -> {*ō} || _ {*r} EnglishSandboxStarConsonant,
    {*a} -> {*ō} || _ {*r} .#.,
    {*a} -> {*ō} || _ {*w}
];

# Late Old English lowering of {*ō} before liquids and in absolute final position.
define EnglishSandboxLiquidLowering [
    {*ō} -> {ɔː} || _ EnglishSandboxStarLiquid,
    {*ō} -> {ɔː} || _ .#.
];

# Anglo-Frisian shortening of {*ō} before velar stops (book/foot class).
define EnglishSandboxVelarShortening [
    {*ō} -> {ʊ} || _ EnglishSandboxStarVelarStop
];

# West Germanic rounding of {*u} before {*r} + consonant or word-final.
define EnglishSandboxUrRounding [
    {*u} -> {ɔː} || _ {*r} EnglishSandboxStarConsonant,
    {*u} -> {ɔː} || _ {*r} .#.
];

# Proto-stage fronting/backing before Middle English developments.
define EnglishSandboxProtoRhoticFronting [
    {*a} -> {æ} || _ {*r}
];

# Short back-vowel lowering (pre-ME contextual rewrite).
define EnglishSandboxShortBackLowering [
    {*o} -> {ɔ}
];

# Deterministic long-ō stage: targets for boot/foot/goose classes.
define EnglishSandboxOFrontContext [
    {*ō} -> {*uː} || _ ({*m} | {*n} | {*l}) EnglishSandboxStarVocalic,
    {*ō} -> {*ʊ} || EnglishSandboxStarConsonant _ EnglishSandboxStarConsonant,
    {*ō} -> {*əʊ}
];

# Deterministic long-ū stage: house vs. oo contexts.
define EnglishSandboxUFrontContext [
    {*ū} -> {*aʊ} || _ ({*s} | {*n} | {*l} | .#.),
    {*ū} -> {*uː}
];

# Anglo-Frisian style short-a fronting in closed syllables.
define EnglishSandboxShortAFronting [
    {*a} -> {*æ} || _ EnglishSandboxStarConsonant
];

# Late Middle English short-vowel splits (KIT vs. DRESS, FOOT vs. STRUT).
define EnglishSandboxShortVowelContextual [
    {u} -> {ʊ} || _ EnglishSandboxStarVelarStop,
    {u} -> {ʊ} || _ {*z} EnglishSandboxStarVocalic,
    {u} -> {ʊ} || _ {*z} EnglishSandboxWeakTailVowel,
    {u} -> {ʊ} || _ {*m} EnglishSandboxStarVocalic,
    {u} -> {ʊ} || _ {*n} EnglishSandboxStarConsonant,
    {u} -> {ʊ} || _ {*l} (EnglishSandboxStarConsonant | .#.),
    {u} -> {ʊ} || _ {l} (EnglishSandboxStarConsonant | .#.),
    {u} -> {ʊ} || _ {*r} (EnglishSandboxStarConsonant | .#.),
    {u} -> {ʊ} || _ {r} (EnglishSandboxStarConsonant | .#.),
    {u} -> {ʊ} || _ ({*f} | {*s} | {*θ}) EnglishSandboxStarConsonant,
    {u} -> {ʊ} || _ ({*f} | {*s} | {*θ}) .#.,
    {u} -> {ʊ} || _ {*t} (EnglishSandboxStarConsonant | .#.),
    {u} -> {ʊ} || _ {t} (EnglishSandboxStarConsonant | .#.),
    {u} -> {ʊ} || _ {*d} (EnglishSandboxStarConsonant | .#.),
    {u} -> {ʊ} || _ {d} (EnglishSandboxStarConsonant | .#.),
    {u} -> {ʊ} || _ {*t} EnglishSandboxWeakTailVowel,
    {u} -> {ʊ} || _ {t} EnglishSandboxWeakTailVowel,
    {u} -> {ʊ} || _ {*d} EnglishSandboxWeakTailVowel,
    {u} -> {ʊ} || _ {d} EnglishSandboxWeakTailVowel,
    {u} -> {ʊ} || _ {r} EnglishSandboxWeakTailVowel,
    {u} -> {ʊ} || _ {*m} EnglishSandboxWeakTailVowel,
    {u} -> {ʊ} || _ {*n} EnglishSandboxWeakTailVowel,
    {e} -> {ɪ} || _ (EnglishSandboxStarNasal | EnglishSandboxStarLiquid) EnglishSandboxStarConsonant,
    {e} -> {ɪ} || _ EnglishSandboxStarNasal .#.,
    {i} -> {ɪ} || _ EnglishSandboxStarConsonant,
    {i} -> {ɪ} || _ .#.
];

define EnglishSandboxShortVowelFallback [
    {u} -> {ʌ},
    {e} -> {ɛ}
];

define EnglishSandboxShortVowelSplit EnglishSandboxShortVowelContextual .o. EnglishSandboxShortVowelFallback;

# Old English & Middle English vowel developments feeding the later English cascade.
define EnglishSandboxCoreVowelRules [
    {*i} -> {i},
    {*u} -> {u}
];

define EnglishSandboxRhoticColoring [
    {*a} EnglishSandboxStarConsonant {*r} -> {*æ} EnglishSandboxStarConsonant {*r},
    {*e} EnglishSandboxStarConsonant {*r} -> {*ɛ} EnglishSandboxStarConsonant {*r},
    {*i} EnglishSandboxStarConsonant {*r} -> {*ə} EnglishSandboxStarConsonant {*r},
    {*u} EnglishSandboxStarConsonant {*r} -> {*ɔː} EnglishSandboxStarConsonant {*r},
    {*o} EnglishSandboxStarConsonant {*r} -> {*ɔː} EnglishSandboxStarConsonant {*r}
];

# Deterministic Great Vowel Shift stages.
define EnglishSandboxGreatVowelShift [
    {*ō} -> {*əʊ},
    {*ā} -> {*eɪ}
];

define EnglishSandboxGVSUShift [
    {*ū} -> {*aʊ}
];

# Late Middle English / Early Modern English unstressed reductions (weak tails).
define EnglishSandboxWeakTailReductions [
    {*a} -> {ə} || _ .#.,
    {*a} -> {ə} || _ EnglishSandboxWeakTailVowel,
    {*ą} -> {*ə}
];

# Early Modern English post-vocalic /r/-loss.
define EnglishSandboxPostVocalicRLoss [
    {*r} -> 0 || EnglishSandboxPlainVocalic _ (EnglishSandboxStarConsonant | .#.)
];

define EnglishSandboxVowelRules EnglishSandboxCoreVowelRules
    .o. EnglishSandboxShortVowelSplit
    .o. EnglishSandboxRhoticColoring
    .o. EnglishSandboxShortBackLowering
    .o. EnglishSandboxOFrontContext
    .o. EnglishSandboxUFrontContext
    .o. EnglishSandboxGreatVowelShift
    .o. EnglishSandboxGVSUShift;

define EnglishSandboxOrthography [
    {*x} -> {h}
];

# Final conversion of starred vowels/diphthongs to IPA (pre-surface cleanup).

    {*ē} -> {iː},
