# A-restoration: */æ/ > */a/ before a back vowel (or AFB-fronted *a) in the following syllable.
# Per Ringe/Taylor Vol 2 §6.3.1: "Those stressed *æ which were immediately followed
# by a single or geminate consonant or sC-cluster which was in turn followed by a
# back vowel became a." Ringe explicitly notes: "That unstressed *æ was not retracted."
#
# CHRONOLOGY: A-restoration applies AFTER AFB (Anglo-Frisian Brightening). Ringe says:
# "retraction must have occurred subsequently to fronting—and subsequently to breaking".
# This means at the time A-restoration applies, underlying PWGmc *a has already been
# fronted to surface *æ by AFB.
#
# RESTORATION TRIGGERS: The rule originally referred to "back vowels" in the following
# syllable. After AFB, the back vowels that remain are {*o}, {*u}, {*ō}, {*ū}, etc.
# BUT: stressed {*a} in endings (like gen.sg. *-as) has been fronted to {*æ} by AFB.
# These surface {*æ} (< underlying {*a}) STILL trigger restoration, because they are
# underlyingly back vowels. This is why gen.sg. forms like *brandas (AFB→ *brændæs)
# trigger restoration: the tail {*æ} is underlyingly {*a} (back).
#
# WEAK-TAIL EXCLUSION: Only FULL (stressed) vowels trigger restoration. Weak-tail
# vowels {*ă} and {*ą} are UNSTRESSED and do NOT trigger restoration, even though
# they are underlyingly back.
# {*ô} (trimoric *ō) is a back vowel that triggers A-restoration, but is not
# in PGmcStarBackVowel to avoid regressions in general vowel rules.
define OEARestorationTriggerVowel [EnglishStarBackVowel | {*æ} | {*ô}];

define OEARestorationIntervening [

# Northwest Germanic u-lowering before non-high vowels (Ringe/Taylor §2.3.1).
# Stressed *u → *o (SHORT o) when next syllable contains non-high vowel (*a, *e, *o, *ā, *ē, *ō)
# and no nasal intervenes in syllable coda. Very early change (pre-OE).
# R/T: "stressed *u was apparently lowered to *o" (p.27, line 2297)
# Examples: *wurda → word, *hurnga → horn, *budanaz → boden (all with SHORT o)
define EnglishStarNonHighVowel [
    {*a} | {*ā} | {*ă} | {*ą} |
    {*e} | {*ē} |
    {*o} | {*ō} | {*ɔ̆}
];

# R/T vol.2 §2.1.1: Stressed *u → *o before non-high vowels in NWGmc,
# EXCEPT when a nasal (*n, *m) follows *u (R/T: "no nasal in the syllable coda,
# nor *j, intervened"). Also blocked before *j (but *j triggers i-umlaut later).
# Northern WGmc (OE/OS) also blocks before intervocalic nasals.
# Exceptions near labials (*full, *wulf, *fugol) are lexically irregular.
define NWGmcULowering [
    {*u} -> {*o} || _ [EnglishStarConsonant - EnglishStarNasal] EnglishStarConsonant* EnglishStarNonHighVowel

# N-stem final -n loss with nasalization (R/T vol.2 §3.1.4, §3.3.1).
# PGmc fem. n-stem nom.sg. *-ōn → *-ǭ (nasalized ō) by loss of final -n.
# The *-ǭ is bimoric; it shortens to *-æ → OE -e (unstressed fronting).
# Masc. n-stem nom.sg. uses trimoric *-ô (no -n), which is NOT affected.
define NWGmcNStemNLoss [
    {*ō} {*n} -> {*ǭ} || _ .#.

# R/T vol.2 p.267: "word-final bimoric non-nasalized long *-ō became

# Examples: PGmc *gebō > PNWGmc *gebu, PGmc *feþrō > PNWGmc *feþru.

# R/T vol.2 §5.1.2: PGmc *ē₁ → OE ǣ. Historically *ē₁ lowered to *ā in NWGmc,

### k-spirantisation and shielding
define GermanStopShiftBackVowel [{*a} | {*ā} | {*ą} | {*o} | {*ō} | {*u} | {*ū} | {*au}];
define GermanStopShiftFrontVowel [{*e} | {*ē} | {*ę} | {*i} | {*ī} | {*y} | {*ai} | {*ei} | {*iu} | {*eu}];

define GermanStopShiftBackLeft [
    GermanStopShiftBackVowel |
    GermanStopShiftBackVowel GermanStopShiftLiquid

define GermanStopShiftFrontLeft [
    GermanStopShiftFrontVowel |
    GermanStopShiftFrontVowel GermanStopShiftLiquid
];

    {*ō} ({*j} | 0) |
