#!/usr/local/bin/foma -f
# Rule-based Proto-Germanic → daughter transducers.

##############################################
## Shared inventory                          ##
##############################################

define ProtoVowel [a|ā|æ|e|ē|i|ī|o|ō|u|ū|ɑ|ɔ|ə|ɛ|ɜ|ʉ|ʊ|ø|œ|y];
define ProtoConsonant [b|d|f|g|h|j|k|l|m|n|p|r|s|t|v|w|x|z|ð|ŋ|ɡ|ɣ|ɪ|ʃ|ʋ|ʌ|ʤ|ʧ];
define ProtoSymbol [ProtoVowel|ProtoConsonant];

# Proto-Germanic templates loosely based on the attested lexicon. This keeps the
# search space finite and filters out wildly ill-formed candidates when inverting
# the daughter rules.
define ProtoNucleus [a|e|i|o|u|ē|ī|ō|ū|{ai}|{au}|{eu}];

define ProtoOnsetCore [
    b|d|f|g|j|k|l|m|n|p|r|s|t|w|x|z|θ|
    bb|bj|bl|bn|br|bs|dj|dm|dr|dw|fl|fn|fr|ft|gj|gl|gn|gr|gw|
    kj|kk|kl|kn|kr|kw|lb|ld|lf|lg|lj|lk|ll|lm|lp|lt|lθ|
    mb|mm|mn|mr|nd|ng|nj|nk|nn|nr|nθ|pn|pp|rb|rd|rg|rk|rm|rn|rp|rr|rt|rw|rx|rθ|
    sk|sl|sm|sn|sp|st|sw|tj|tn|tr|tt|wj|wr|ww|
    xm|xn|xr|xs|xt|xw|zd|zj|zn|zw|θl|θr|
    kkj|ldr|lgw|llj|ltj|ndj|ndl|ndr|ngj|ngr|ngw|ntr|rst|rwj|rxt|spr|stj|str|wwj|xtj|nxwst
];
define ProtoOnset [0|ProtoOnsetCore];

define ProtoCodaCore [
    b|d|f|g|j|k|l|m|n|p|r|s|t|w|x|z|θ|
    bb|bj|bl|bn|br|bs|dj|dm|dr|dw|fl|fn|ft|gj|gl|gn|gw|
    kj|kk|kn|kz|lb|ld|lf|lg|lj|lk|ll|lm|lp|lt|lθ|
    mb|mm|mn|mr|nd|ng|nj|nk|nn|nr|nθ|pn|pp|rb|rd|rg|rk|rm|rn|rp|rr|rt|rw|rx|rθ|
    sk|sl|sm|st|sz|tj|tn|tr|tt|wj|ww|wz|
    xm|xs|xt|xw|xz|zd|zj|zn|zw|θl|θr|θz|
    kkj|ldr|lgw|llj|ltj|ndj|ndl|ndr|ndz|ngj|ngr|ngw|nsz|ntr|rst|rwj|rxt|stj|stz|wwj|xtj|xtz|nxwst
];
define ProtoCoda [0|ProtoCodaCore];

define ProtoSyllable ProtoOnset ProtoNucleus ProtoCoda;
define ProtoWord ProtoSyllable+;

define Cons ProtoConsonant;
define Vowel ProtoVowel;
define FrontVowel [i|ī|e|ē|æ|ɛ|ø|œ|y|ʉ];
define BackVowel [a|ā|ɑ|o|ō|u|ū|ɔ|ʊ];
define Sonorant [j|r|l|m|n|ŋ];

##############################################
## Surface phonotactics (baggy filters)     ##
##############################################

# English allows sizable clusters, but every syllable must anchor to a vowel.
define EnglishSurfaceVowel [
    {aɪ} | {aʊ} | {eɪ} | {iː} | {uː} | {æ} | {ɑː} | {ɔ} | {ɔː} |
    {ə} | {əʊ} | {ɛ} | {ɛə} | {ɜː} | {ɪ} | {ɪə} | {ʊ} | {ʌ}
];
define EnglishSurfaceConsonant [
    {b} | {d} | {f} | {h} | {j} | {k} | {l} | {m} | {n} | {p} |
    {r} | {s} | {t} | {v} | {z} | {ð} | {ŋ} | {ɡ} | {ʃ} | {ʋ} |
    {ʤ} | {ʧ} | {θ}
];
define EnglishSurfaceOnset [
    0 |
    EnglishSurfaceConsonant |
    EnglishSurfaceConsonant EnglishSurfaceConsonant |
    EnglishSurfaceConsonant EnglishSurfaceConsonant EnglishSurfaceConsonant
];
define EnglishSurfaceCoda [
    0 |
    EnglishSurfaceConsonant |
    EnglishSurfaceConsonant EnglishSurfaceConsonant |
    EnglishSurfaceConsonant EnglishSurfaceConsonant EnglishSurfaceConsonant
];
define EnglishSurfaceSyllable EnglishSurfaceOnset EnglishSurfaceVowel EnglishSurfaceCoda;
define EnglishSurface EnglishSurfaceSyllable+;

# Dutch surface strings share the same "at least one vowel" requirement.
define DutchSurfaceVowel [
    {aː} | {eː} | {iː} | {oː} | {uː} | {yː} | {øː} | {ɑ} | {ɑu} |
    {ɔ} | {ə} | {ɛ} | {ɛɪ} | {ɪ} | {ʉ} | {œy}
];
define DutchSurfaceConsonant [
    {b} | {d} | {f} | {h} | {j} | {k} | {l} | {m} | {n} | {p} |
    {r} | {s} | {t} | {v} | {x} | {z} | {ŋ} | {ɣ} | {ʋ}
];
define DutchSurfaceOnset [
    0 |
    DutchSurfaceConsonant |
    DutchSurfaceConsonant DutchSurfaceConsonant |
    DutchSurfaceConsonant DutchSurfaceConsonant DutchSurfaceConsonant
];
define DutchSurfaceCoda [
    0 |
    DutchSurfaceConsonant |
    DutchSurfaceConsonant DutchSurfaceConsonant |
    DutchSurfaceConsonant DutchSurfaceConsonant DutchSurfaceConsonant
];
define DutchSurfaceSyllable DutchSurfaceOnset DutchSurfaceVowel DutchSurfaceCoda;
define DutchSurface DutchSurfaceSyllable+;

# German admits a few affricates and long vowels; keep the filter permissive.
define GermanSurfaceVowel [
    {a} | {ai} | {au} | {aː} | {eː} | {iː} | {oː} | {uː} |
    {ɔ} | {ɔy} | {ə} | {ɛ} | {ɪ} | {ʊ}
];
define GermanSurfaceConsonant [
    {b} | {d} | {d/t} | {f} | {h} | {j} | {k} | {l} | {m} | {n} |
    {p} | {pf} | {r} | {s} | {t} | {ts} | {v} | {x} | {z} | {ŋ} |
    {ɡ} | {ʃ}
];
define GermanSurfaceOnset [
    0 |
    GermanSurfaceConsonant |
    GermanSurfaceConsonant GermanSurfaceConsonant |
    GermanSurfaceConsonant GermanSurfaceConsonant GermanSurfaceConsonant
];
define GermanSurfaceCoda [
    0 |
    GermanSurfaceConsonant |
    GermanSurfaceConsonant GermanSurfaceConsonant |
    GermanSurfaceConsonant GermanSurfaceConsonant GermanSurfaceConsonant
];
define GermanSurfaceSyllable GermanSurfaceOnset GermanSurfaceVowel GermanSurfaceCoda;
define GermanSurface GermanSurfaceSyllable+;

##############################################
## English rules                             ##
##############################################

define EnglishRules [
    "x" -> "h",
    "w" -> "ʋ",
    "w" -> 0 || Vowel _ Vowel,
    "z" -> "r" || Vowel _ Vowel,
    "z" -> 0 || _ .#.,
    "dj" -> "nd",
    "ē" -> "iː",
    "ō" -> "uː",
    "au" -> "oː",
    "ai" -> "eɪ",
    "e" -> "ɪ" || _ Cons,
    "a" -> "ə" || _ .#.,
    "a" -> "æ" || _ "r",
    "u" -> "ʌ" || _ Cons
];

define English ProtoWord .o. EnglishRules .o. EnglishSurface;

clear stack
regex English;
save stack english.bin

##############################################
## Dutch rules                               ##
##############################################

define DutchRules [
    "x" -> "h",
    "w" -> "ʋ",
    "v" -> "f" || .# _ ,
    "z" -> "r" || Vowel _ Vowel,
    "z" -> 0 || _ .#.,
    "dj" -> "nd",
    "ō" -> "oː",
    "ē" -> "eː",
    "au" -> "œy",
    "ai" -> "ɛi",
    "u" -> "ɔ" || _ .#.,
    "a" -> "ɑ",
    "a" -> "ə" || _ Cons,
    "n" -> "ə" || _ .#.,
    "e" -> "ɪ" || _ Cons,
    "g" -> "x"
];

define Dutch ProtoWord .o. DutchRules .o. DutchSurface;

clear stack
regex Dutch;
save stack dutch.bin

##############################################
## German rules                              ##
##############################################

define GermanCoreRules [
    "p" -> "f" || .# _ ,
    "t" -> "ts" || .# _ ,
    "s" -> "ʃ" || .# _ ,
    "w" -> "v",
    "x" -> "h",
    "z" -> "r" || Vowel _ Vowel,
    "z" -> 0 || _ .#.,
    "w" -> 0 || Vowel _ Vowel,
    "u" -> "ɔ" || _ ("k" | "K"),
    "dj" -> "nd",
    "u" -> "ɔ" || _ .#.,
    "n" -> "ə" || _ Cons,
    "a" -> "ə" || _ Cons,
    "e" -> "ɪ" || _ Cons,
    "s" -> "z" || Vowel _ Vowel,
    "θ" -> "d"
];

define GermanStopShift [
    "kk" -> "K",
    "k" -> "x" || BackVowel _ (Cons | .#.),
    "k" -> "ç" || FrontVowel _ (Cons | .#.)
];

define GermanCleanup [
    "K" -> "k"
];

define GermanRules GermanCoreRules .o. GermanStopShift .o. GermanCleanup;

define German ProtoWord .o. GermanRules .o. GermanSurface;

clear stack
regex German;
save stack german.bin
